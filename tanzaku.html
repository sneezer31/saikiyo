// 2. テキスト描画設定 const rawText = textInput.value; const lines = rawText.split('\n'); // 文字サイズの縮小と、組版比率の正常化 const fontSize = 34; // バシッと収まるサイズ const charStepY = fontSize * 1.15; // 縦方向の文字送り（文字間を締める） const lineStepX = fontSize * 1.8; // 横方向の行送り（行間を適度に空ける） ctx.font = `500 ${fontSize}px ${fontSelect.value}`; ctx.fillStyle = colorText.value; ctx.textBaseline = 'top'; ctx.textAlign = 'center'; const strokeMode = strokeSelect.value; if (strokeMode === 'white') { ctx.strokeStyle = '#FFFFFF'; ctx.lineWidth = 6; ctx.lineJoin = 'round'; } else if (strokeMode === 'gray') { ctx.strokeStyle = '#888888'; ctx.lineWidth = 6; ctx.lineJoin = 'round'; } // --- 完全自動センタリングのための事前計算 --- let maxChars = 0; let blockWidthOffset = 0; for (let i = 0; i < lines.length; i++) { const line = lines[i]; if (line === "") { // 空行は0.5行として計算 blockWidthOffset += lineStepX * 0.5; } else { // 通常の行 if (i !== 0) blockWidthOffset += lineStepX; if (line.length > maxChars) maxChars = line.length; } } // ブロック全体の面積を算出し、キャンバス中央へのオフセットを決定 const blockHeight = maxChars * charStepY; const blockWidth = blockWidthOffset + fontSize; // Y座標：最大行長から高さを算出し、上下中央へ（最低50pxの余白は担保） const startY = Math.max(50, (height - blockHeight) / 2); // X座標：ブロック全体の幅を算出し、左右中央へ（右端を開始位置とする） const startX = (width + blockWidth) / 2 - fontSize / 2; let currentX = startX; for (let i = 0; i < lines.length; i++) { const line = lines[i]; // 空行（0.5行スキップ）の実行 if (line === "") { currentX -= lineStepX * 0.5; continue; } // 天揃えのため、全行同じstartYから開始 let currentY = startY; for (let j = 0; j < line.length; j++) { const char = line[j]; // 縦書き用の文字置換 let drawChar = char; if (char === '「' || char === '『' || char === '（') drawChar = '︵'; else if (char === '」' || char === '』' || char === '）') drawChar = '︶'; else if (char === 'ー' || char === '―') drawChar = '｜'; // 描画実行 if (strokeMode !== 'none') { ctx.strokeText(drawChar, currentX, currentY); } ctx.fillText(drawChar, currentX, currentY); currentY += charStepY; } // 次の列（左へ移動） currentX -= lineStepX; } }
